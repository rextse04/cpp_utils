838 size_type level_idx, left_idx, right_idx, begin, end, div_node;

841 constexpr auto do_traverse(this auto&& self, const auto& traverser, const auto& strategy,
842 int level, size_type level_idx, size_type begin, size_type end) {
843 using enum interval_endpoint_type;
844 node_info info;
845 info.level = level;
846 info.level_idx = level_idx;
847 info.begin = begin;
848 info.end = end;
849 info.div_node = begin + (std::bit_ceil(end - begin) >> 1);
850 info.left_level = level - 1;
851 info.left_idx = level_idx * 2;
852 info.right_level = std::bit_width(end - info.div_node - 1);
853 info.right_idx = ((level_idx * 2 + 1) << (level - info.right_level)) >> 1;
854 const node_info* info_ptr = std::addressof(info);
855 const segment_tree_traverse_direction dir = traverser(self, info);
856 if (dir == stop) return strategy(self, info_ptr, nullptr, nullptr);
857 if (level == 0) return strategy(self, nullptr, nullptr, nullptr);
858 if constexpr (requires { strategy.prepare(self, *info_ptr); }) strategy.prepare(self, *info_ptr);
859 const auto collect_left = [&] {
860 return self.do_traverse(traverser, strategy, info.left_level, info.left_idx, info.begin, info.div_node);
861 };
862 const auto collect_right = [&] {
863 return self.do_traverse(traverser, strategy, info.right_level, info.right_idx, info.div_node, info.end);
864 };
865 if (dir == left) {
866 const auto& left = collect_left();
867 return strategy(self, info_ptr, std::addressof(left), nullptr);
868 }
869 if (dir == right) {
870 const auto& right = collect_right();
871 return strategy(self, info_ptr, nullptr, std::addressof(right));
872 }
873 if (dir == both) {
874 const auto& left = collect_left();
875 const auto& right = collect_right();
876 return strategy(self, info_ptr, std::addressof(left), std::addressof(right));
877 }
878 std::unreachable();
