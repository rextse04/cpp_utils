498 template <typename T, typename S>
499 constexpr auto scalar_shift_left::operator()(T&& t, S&& s) noexcept
500 -> decltype(std::declval<T&&>() << std::declval<S&&>()) {
501 if (s < 0) return scalar_shift_right{}(std::forward<T>(t), -std::forward<S>(s));
502 if (s >= width_of_v<std::remove_cvref_t<T>>) return 0;
503 return std::forward<T>(t) << std::forward<S>(s);
504 }

506 constexpr auto scalar_shift_right::operator()(T&& t, S&& s) noexcept
507 -> decltype(std::declval<T&&>() >> std::declval<S&&>()) {
508 using TD = std::remove_cvref_t<T>;
509 if (s < 0) return scalar_shift_left{}(std::forward<T>(t), -std::forward<S>(s));
510 if (s >= width_of_v<TD>) {
511 if constexpr (std::numeric_limits<TD>::is_signed) return -1;
512 else return 0;
513 }
514 return std::forward<T>(t) >> std::forward<S>(s);
515 }
