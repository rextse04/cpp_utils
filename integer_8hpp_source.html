492 template <typename T, typename S>
493 constexpr auto scalar_shift_left::operator()(T&& t, S&& s) noexcept
494 -> decltype(std::declval<T&&>() << std::declval<S&&>()) {
495 if (s < 0) return scalar_shift_right{}(std::forward<T>(t), -std::forward<S>(s));
496 if (s >= width_of_v<std::remove_cvref_t<T>>) return 0;
497 return std::forward<T>(t) << std::forward<S>(s);
498 }
499 template <typename T, typename S>
500 constexpr auto scalar_shift_right::operator()(T&& t, S&& s) noexcept
501 -> decltype(std::declval<T&&>() >> std::declval<S&&>()) {
502 using TD = std::remove_cvref_t<T>;
503 if (s < 0) return scalar_shift_left{}(std::forward<T>(t), -std::forward<S>(s));
504 if (s >= width_of_v<TD>) {
505 if constexpr (std::numeric_limits<TD>::is_signed) return -1;
506 else return 0;
507 }
508 return std::forward<T>(t) >> std::forward<S>(s);
509 }
