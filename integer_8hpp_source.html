503 template <typename T, typename S>
504 constexpr auto scalar_shift_left::operator()(T&& t, S&& s) noexcept
505 -> decltype(std::declval<T&&>() << std::declval<S&&>()) {
506 if (s < 0) return scalar_shift_right{}(std::forward<T>(t), -std::forward<S>(s));

508 return std::forward<T>(t) << std::forward<S>(s);

512 -> decltype(std::declval<T&&>() >> std::declval<S&&>()) {
513 using TD = std::remove_cvref_t<T>;
514 if (s < 0) return scalar_shift_left{}(std::forward<T>(t), -std::forward<S>(s));
515 if (s >= width_of_v<TD>) {
516 if constexpr (std::numeric_limits<TD>::is_signed) return -1;
517 else return 0;
518 }
519 return std::forward<T>(t) >> std::forward<S>(s);
