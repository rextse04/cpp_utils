481 -> decltype(std::declval<T&&>() << std::declval<S&&>()) {

484 return std::forward<T>(t) << std::forward<S>(s);

488 -> decltype(std::declval<T&&>() >> std::declval<S&&>()) {
489 using TD = std::remove_cvref_t<T>;
490 if (s < 0) return scalar_shift_left{}(std::forward<T>(t), -std::forward<S>(s));
491 if (s >= width_of_v<TD>) {
492 if constexpr (std::numeric_limits<TD>::is_signed) return -1;
493 else return 0;
494 }
